## Usage

```tsx
import { AspectRatio } from "@/components/ui/aspect-ratio"
```

```tsx
<AspectRatio ratio={16 / 9} className="overflow-hidden rounded-xl">
  <img
    src="..."
    alt="Image"
    className="absolute inset-0 size-full object-cover"
  />
</AspectRatio>
```
